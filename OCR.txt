package vn.edu.usth.tip.utils;

import java.util.regex.Matcher;
import java.util.regex.Pattern;

public final class InvoiceParser {

    // Lines that signal a total amount (Vietnamese + English)
    private static final Pattern TOTAL_LINE = Pattern.compile(
        "(?i)(?:tổng|total|thành tiền|cộng|thanh toán)[^\\d]*(\\d[\\d.,]+)",
        Pattern.UNICODE_CASE
    );

    // Any number followed by a currency symbol
    private static final Pattern MONEY_AMOUNT = Pattern.compile(
        "(\\d[\\d.,]+)\\s*(?:đ|₫|VND|VNĐ)",
        Pattern.CASE_INSENSITIVE
    );

    /** Returns the best-guess total amount in VND (0 if not found). */
    public static long parseAmount(String text) {
        if (text == null) return 0;

        // Pass 1: lines that explicitly say "tổng / total / thành tiền"
        Matcher m = TOTAL_LINE.matcher(text);
        long best = 0;
        while (m.find()) {
            long v = toNumber(m.group(1));
            if (v > best) best = v;
        }
        if (best > 0) return best;

        // Pass 2: largest explicit VND amount anywhere in the text
        m = MONEY_AMOUNT.matcher(text);
        while (m.find()) {
            long v = toNumber(m.group(1));
            if (v > best) best = v;
        }
        return best;
    }

    /**
     * Returns a short description line (vendor / store name).
     * Strategy: first non-numeric, non-trivial line of OCR text.
     */
    public static String parseNote(String text) {
        if (text == null || text.trim().isEmpty()) return "";
        for (String line : text.split("\n")) {
            line = line.trim();
            if (line.length() > 2 && !line.matches("[\\d\\s.,:/\\-]+")) {
                return line.length() > 80 ? line.substring(0, 80) : line;
            }
        }
        return "";
    }

    /** Strip thousands separators (both ',' and '.') then parse as long. */
    private static long toNumber(String s) {
        try {
            return Long.parseLong(s.replaceAll("[.,]", ""));
        } catch (NumberFormatException e) {
            return 0;
        }
    }

    private InvoiceParser() {}
}

package vn.edu.usth.tip.ui.fragments;

import android.Manifest;
import android.content.pm.PackageManager;
import android.net.Uri;
import android.os.Bundle;
import android.view.*;
import android.widget.Toast;

import androidx.activity.result.ActivityResultLauncher;
import androidx.activity.result.contract.ActivityResultContracts;
import androidx.annotation.*;
import androidx.camera.core.*;
import androidx.camera.lifecycle.ProcessCameraProvider;
import androidx.camera.view.PreviewView;
import androidx.core.content.ContextCompat;
import androidx.fragment.app.Fragment;
import androidx.navigation.Navigation;

import com.google.common.util.concurrent.ListenableFuture;
import com.google.mlkit.vision.common.InputImage;
import com.google.mlkit.vision.text.TextRecognition;
import com.google.mlkit.vision.text.TextRecognizer;
import com.google.mlkit.vision.text.latin.TextRecognizerOptions;

import java.io.File;
import java.io.IOException;

import vn.edu.usth.tip.R;
import vn.edu.usth.tip.utils.InvoiceParser;

public class ScanReceiptFragment extends Fragment {

    // ── Bundle keys shared with ExtractInvoiceFragment ──────────────────────
    public static final String KEY_AMOUNT = "ocr_amount";
    public static final String KEY_NOTE   = "ocr_note";

    // ── Views ────────────────────────────────────────────────────────────────
    private PreviewView previewView;
    private View layoutLoading, scanLaser;

    // ── CameraX ──────────────────────────────────────────────────────────────
    private ImageCapture imageCapture;
    private Camera camera;

    // ── ActivityResult launchers (MUST register in onCreate) ─────────────────
    private ActivityResultLauncher<String> cameraPermLauncher;
    private ActivityResultLauncher<String> galleryLauncher;

    public ScanReceiptFragment() {}

    @Override
    public void onCreate(@Nullable Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        cameraPermLauncher = registerForActivityResult(
            new ActivityResultContracts.RequestPermission(),
            granted -> {
                if (granted) startCamera();
                else Toast.makeText(requireContext(),
                        "Cần quyền camera để quét hóa đơn", Toast.LENGTH_LONG).show();
            }
        );

        galleryLauncher = registerForActivityResult(
            new ActivityResultContracts.GetContent(),
            uri -> { if (uri != null) processUri(uri); }
        );
    }

    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, ViewGroup container,
                             Bundle savedInstanceState) {
        return inflater.inflate(R.layout.fragment_scan_receipt, container, false);
    }

    @Override
    public void onViewCreated(@NonNull View view, @Nullable Bundle savedInstanceState) {
        super.onViewCreated(view, savedInstanceState);

        previewView   = view.findViewById(R.id.preview_view);
        layoutLoading = view.findViewById(R.id.layout_loading);
        scanLaser     = view.findViewById(R.id.scan_laser);

        // Laser animation (unchanged from original)
        if (scanLaser != null) {
            android.view.animation.TranslateAnimation anim = new android.view.animation.TranslateAnimation(
                    android.view.animation.Animation.RELATIVE_TO_PARENT, 0f,
                    android.view.animation.Animation.RELATIVE_TO_PARENT, 0f,
                    android.view.animation.Animation.RELATIVE_TO_PARENT, -0.2f,
                    android.view.animation.Animation.RELATIVE_TO_PARENT, 0.7f);
            anim.setDuration(1500);
            anim.setRepeatCount(android.view.animation.Animation.INFINITE);
            anim.setRepeatMode(android.view.animation.Animation.REVERSE);
            scanLaser.startAnimation(anim);
        }

        view.findViewById(R.id.btn_back).setOnClickListener(
                v -> Navigation.findNavController(v).navigateUp());

        view.findViewById(R.id.btn_flash).setOnClickListener(v -> toggleFlash());

        view.findViewById(R.id.btn_gallery).setOnClickListener(
                v -> galleryLauncher.launch("image/*"));

        view.findViewById(R.id.btn_capture).setOnClickListener(v -> capturePhoto());

        // Request camera permission or start directly
        if (ContextCompat.checkSelfPermission(requireContext(), Manifest.permission.CAMERA)
                == PackageManager.PERMISSION_GRANTED) {
            startCamera();
        } else {
            cameraPermLauncher.launch(Manifest.permission.CAMERA);
        }
    }

    // ── CameraX ──────────────────────────────────────────────────────────────

    private void startCamera() {
        ListenableFuture<ProcessCameraProvider> future =
                ProcessCameraProvider.getInstance(requireContext());
        future.addListener(() -> {
            try {
                ProcessCameraProvider provider = future.get();

                Preview preview = new Preview.Builder().build();
                preview.setSurfaceProvider(previewView.getSurfaceProvider());

                imageCapture = new ImageCapture.Builder()
                        .setCaptureMode(ImageCapture.CAPTURE_MODE_MINIMIZE_LATENCY)
                        .build();

                provider.unbindAll();
                camera = provider.bindToLifecycle(
                        this, CameraSelector.DEFAULT_BACK_CAMERA, preview, imageCapture);
            } catch (Exception e) {
                Toast.makeText(requireContext(), "Không mở được camera", Toast.LENGTH_SHORT).show();
            }
        }, ContextCompat.getMainExecutor(requireContext()));
    }

    private void capturePhoto() {
        if (imageCapture == null) return;
        showLoading(true);

        File photoFile = new File(requireContext().getCacheDir(),
                "receipt_" + System.currentTimeMillis() + ".jpg");
        ImageCapture.OutputFileOptions opts =
                new ImageCapture.OutputFileOptions.Builder(photoFile).build();

        imageCapture.takePicture(opts, ContextCompat.getMainExecutor(requireContext()),
                new ImageCapture.OnImageSavedCallback() {
                    @Override
                    public void onImageSaved(@NonNull ImageCapture.OutputFileResults r) {
                        processUri(Uri.fromFile(photoFile));
                    }
                    @Override
                    public void onError(@NonNull ImageCaptureException e) {
                        showLoading(false);
                        Toast.makeText(requireContext(),
                                "Chụp ảnh thất bại", Toast.LENGTH_SHORT).show();
                    }
                });
    }

    private void toggleFlash() {
        if (camera == null) return;
        boolean current = Boolean.TRUE.equals(
                camera.getCameraInfo().getTorchState().getValue() ==
                        androidx.camera.core.TorchState.ON);
        camera.getCameraControl().enableTorch(!current);
    }

    // ── OCR ──────────────────────────────────────────────────────────────────

    private void processUri(Uri uri) {
        showLoading(true);
        try {
            InputImage image = InputImage.fromFilePath(requireContext(), uri);
            TextRecognizer recognizer =
                    TextRecognition.getClient(TextRecognizerOptions.DEFAULT_OPTIONS);
            recognizer.process(image)
                    .addOnSuccessListener(result -> {
                        String raw = result.getText();
                        long amount = InvoiceParser.parseAmount(raw);
                        String note = InvoiceParser.parseNote(raw);
                        navigateToExtract(amount, note);
                    })
                    .addOnFailureListener(e -> {
                        showLoading(false);
                        Toast.makeText(requireContext(),
                                "Không đọc được văn bản trên hóa đơn", Toast.LENGTH_SHORT).show();
                    });
        } catch (IOException e) {
            showLoading(false);
            Toast.makeText(requireContext(), "Lỗi đọc ảnh", Toast.LENGTH_SHORT).show();
        }
    }

    private void navigateToExtract(long amount, String note) {
        if (!isAdded() || getView() == null) return;
        Bundle args = new Bundle();
        args.putLong(KEY_AMOUNT, amount);
        args.putString(KEY_NOTE, note);
        showLoading(false);
        Navigation.findNavController(requireView())
                .navigate(R.id.action_scanReceipt_to_extractInvoice, args);
    }

    private void showLoading(boolean show) {
        if (layoutLoading != null)
            layoutLoading.setVisibility(show ? View.VISIBLE : View.GONE);
    }
}
